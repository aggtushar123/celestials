# UI Impact Surface Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Move the Impact surface (`/segments`, `/surveys`, `/scenarios` with its rules, `/projects`, `/campaigns`, `/canvas`) onto the shared UI primitives in `src/components/ui/`: every route titled by its page through `PageHeader` in the Navbar bar, one `ListCard` per list with rows on `ListRow` (figures in aligned columns from a container breakpoint), one `Toolbar` per list page, `Tabs` / `SegmentedControl` / `SearchInput` / `Select` each in its own role, detail pages on one `DetailHeader` (two visible actions plus a `⋯` menu with Delete in the danger tone), builder modals flattened (no box inside the dialog), loading, empty and error states inside their cards, the four-size type scale, figures in the body font, and colour only where it means status. - 44px targets below `sm` (`min-h-11`); one primary per form, toolbar or page bar; a `role="status"` region is never `display:none`, never `empty:hidden`, and never unmounted between updates: the element is always rendered and only its children change (`sr-only` or `empty:sr-only` when it carries spacing); `min-h-[100dvh]`, never `h-screen`; Lucide icons only, `aria-hidden="true"` when decorative; every icon-only button has an `aria-label`.
- Hover-revealed row actions use `ROW_ACTIONS` (`ui/List.tsx`): opacity only, so they stay in the tab order, and always visible on touch (`[@media(hover:none)]:opacity-100`).

| `font-mono-brand` / `MONO` on words or initials ("NPS", "Day", a type chip) | removed (body font) |
| `font-bold`, `font-extrabold`, `font-medium` | `font-semibold` on headings, labels, buttons; removed on body text |
| `text-[9.5px]`, `[10px]`, `[10.5px]`, `[11.5px]`, `text-xs` | `text-[11px]` (`TYPE.meta`) |
| `text-[12px]` | `13` on controls, inputs, body; `11` on captions, helper text, badges |
| `text-[12.5px]`, `[13.5px]`, `[14px]`, `[14.5px]`, `text-sm`, `text-base` | `text-[13px]` (`TYPE.body`/`TYPE.bodyMuted`); a heading that used 14 becomes `TYPE.title` |
| `text-[16px]`..`[18px]`, `[16.5px]`, `text-lg` | `TYPE.title` |
| `text-[20px]`, `[23px]`, `[24px]`, `text-xl`, `text-2xl` | `TYPE.display` |

| A meta-line link `inline-flex min-h-11 items-center … sm:min-h-0` | `` `${INLINE_TARGET} z-10 …` `` with the meta's own size and colour (`text-ink hover:underline`) |
| `'—'` written in a line that changes | `NO_VALUE` (`ui/format.ts`) |

| `transition-colors`, `transition-all`, `transition-shadow` on rows, chips and buttons | removed (no motion on high-frequency hover); a disclosure chevron's `rotate-90` stays (no transition) |

```tsx
/** One card holding a labelled list. `state` (a LoadingState, EmptyState
 *  or ErrorState) takes the list's place inside the same card. `footer`
 *  (Show more) is a divided row inside the card, drawn only when given;
 *  `busy` marks the list while a new query reads. */
export function ListCard({
  label,
  header,
  state,
  footer,
  busy = false,
  className = '',
  children,
}: {
  label: string;
  header?: ReactNode;
  state?: ReactNode;
  footer?: ReactNode;
  busy?: boolean;
  className?: string;
  children?: ReactNode;
}) {
  return (
    <Card pad="none" className={`overflow-hidden ${className}`.trim()}>
      {header}
      {state ? (
        state
      ) : (
        <ul role="list" aria-label={label} aria-busy={busy || undefined} className="divide-y divide-line-subtle">
          {children}
        </ul>
      )}
      {footer ? (
        <div data-part="list-footer" className="flex flex-col items-center gap-1 border-t border-line-subtle px-4 py-2">
          {footer}
        </div>
      ) : null}
    </Card>
  );
}
```

```tsx
import { useEffect, useId } from 'react';
import { ChevronDown } from 'lucide-react';
import { INVALID, SELECT } from './styles';
import { TYPE } from './type';

export interface SelectOption<T extends string> {
  value: T;
  label: string;
  /** Shown but not choosable (a plan's Done before every milestone is). */
  disabled?: boolean;
}

/** Options under one `<optgroup>` label ("AI attributes"). */
export interface SelectGroup<T extends string> {
  label: string;
  options: readonly SelectOption<T>[];
}

function optionOf<T extends string>(option: SelectOption<T>) {
  return (
    <option key={option.value} value={option.value} disabled={option.disabled}>
      {option.label}
    </option>
  );
}

/** A native select with the house frame and chevron (UI foundations spec
 *  §5.5). The label sits `above` (forms), `inline` to the left ("Sort by",
 *  toolbars), `hidden` (a control whose context names it) or `external`
 *  (no label of its own: a `<label htmlFor={id}>` elsewhere, the rules'
 *  `Field`, names it, so `id` is required). An inline select never shrinks
 *  below its own width: the row wraps instead (Pipelines' Sort by was
 *  clipped beside the Ask rail). `invalid` sets `aria-invalid` and the
 *  danger edge; `describedBy` points at its hint and error; `className` is
 *  layout only (a width, `self-start`), on the outer wrapper. */
export function Select<T extends string>({
  label,
  labelPosition = 'above',
  value,
  onChange,
  options,
  disabled = false,
  id,
  invalid = false,
  describedBy,
  className = '',
}: {
  label: string;
  labelPosition?: 'above' | 'inline' | 'hidden' | 'external';
  value: T;
  onChange: (value: T) => void;
  options: readonly (SelectOption<T> | SelectGroup<T>)[];
  disabled?: boolean;
  id?: string;
  invalid?: boolean;
  describedBy?: string;
  className?: string;
}) {
  const ownId = useId();
  const selectId = id ?? ownId;
  useEffect(() => {
    if (import.meta.env.DEV && labelPosition === 'external' && !id) console.error('Select: labelPosition="external" needs the id its label points at');
  }, [labelPosition, id]);
  const control = (
    <div className={`relative min-w-0 ${labelPosition === 'external' ? className : ''}`.trim()}>
      <select
        id={selectId}
        value={value}
        disabled={disabled}
        aria-invalid={invalid || undefined}
        aria-describedby={describedBy}
        onChange={(event) => onChange(event.target.value as T)}
        className={`${SELECT} ${INVALID}`}
      >
        {options.map((item) =>
          'options' in item ? (
            <optgroup key={`group-${item.label}`} label={item.label}>
              {item.options.map(optionOf)}
            </optgroup>
          ) : (
            optionOf(item)
          ),
        )}
      </select>
      <ChevronDown className="pointer-events-none absolute right-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-ink-muted" aria-hidden="true" />
    </div>
  );
  if (labelPosition === 'external') return control;
  if (labelPosition === 'inline') {
    return (
      <div className={`flex shrink-0 items-center gap-2 ${className}`.trim()}>
        <label htmlFor={selectId} className={`shrink-0 ${TYPE.bodyMuted}`}>
          {label}
        </label>
        {control}
      </div>
    );
  }
  return (
    <div className={`flex flex-col gap-1.5 ${className}`.trim()}>
      <label htmlFor={selectId} className={labelPosition === 'hidden' ? 'sr-only' : TYPE.label}>
        {label}
      </label>
      {control}
    </div>
  );
}
```

```tsx
import { useEffect, type ReactNode, type Ref } from 'react';
import { Link } from 'react-router-dom';
import { Ellipsis, type LucideIcon } from 'lucide-react';
import { Menu, type MenuItem } from './Menu';
import { BUTTON, FOCUS, ICON_BUTTON, PRIMARY } from './styles';
import { TYPE } from './type';

/** One visible action on a detail page's name row. */
export interface HeaderAction {
  key: string;
  label: string;
  icon: LucideIcon;
  /** A link (Edit opens its route over the page). */
  to?: string;
  /** A button. */
  onSelect?: () => void;
  /** The page's one primary (Send now, Save). */
  primary?: boolean;
  disabled?: boolean;
  /** A toggle (Edit, Done editing). */
  pressed?: boolean;
  /** Its words while it works ("Saving…"); also marks it aria-busy. */
  busyLabel?: string;
}

const ACTION = 'min-w-11 justify-center sm:min-w-0';

function ActionButton({ action }: { action: HeaderAction }) {
  const Icon = action.icon;
  const className = `${action.primary ? PRIMARY : BUTTON} ${ACTION}`;
  const body = (
    <>
      <Icon className="h-4 w-4" aria-hidden="true" />
      {/* Icon-only below sm, so the name keeps the row. */}
      <span className="sr-only sm:not-sr-only">{action.busyLabel ?? action.label}</span>
    </>
  );
  if (action.to && !action.disabled) {
    return (
      <Link to={action.to} className={className}>
        {body}
      </Link>
    );
  }
  return (
    <button
      type="button"
      onClick={action.onSelect}
      disabled={action.disabled}
      aria-pressed={action.pressed}
      aria-busy={action.busyLabel ? true : undefined}
      className={className}
    >
      {body}
    </button>
  );
}

/** A detail page's name row (UI foundations spec §6.1, §7, §9.2 row 3; the
 *  Impact pages): the name as the page's one display h1 (or `heading`, an
 *  editable name that holds its own h1), a short `description`, then on the
 *  right at most two visible controls (`control`, such as a rule's switch or
 *  a plan's status, counts as one) and the ⋯ menu, whose danger items
 *  (Delete, Cancel plan) always come last. Under it one muted `meta` line,
 *  then a polite `status` region that is always mounted (empty until there
 *  is something to say: Unsaved changes, Sending now, Paused), then
 *  `children` (a rule sentence, Made by). Below sm the actions are icons
 *  with their names for screen readers, so the row never wraps. */
export function DetailHeader({
  title,
  titleRef,
  heading,
  description,
  control,
  actions = [],
  menu = [],
  menuLabel = 'More actions',
  meta,
  status,
  children,
}: {
  title: string;
  /** The h1, focusable (`tabIndex -1`) when given: Duplicate lands here. */
  titleRef?: Ref<HTMLHeadingElement>;
  /** Replaces the h1 (an editable name); it must render the page's h1. */
  heading?: ReactNode;
  description?: ReactNode;
  /** One always-visible control before the actions. */
  control?: ReactNode;
  /** At most two, a `control` counting as one; more logs a development
   *  error (the test pins it). */
  actions?: readonly HeaderAction[];
  menu?: MenuItem[];
  menuLabel?: string;
  meta?: ReactNode;
  status?: ReactNode;
  children?: ReactNode;
}) {
  const visible = (control ? 1 : 0) + actions.length;
  useEffect(() => {
    if (import.meta.env.DEV && visible > 2) console.error('DetailHeader: at most two visible controls');
  }, [visible]);
  const items = [...menu.filter((item) => !item.danger), ...menu.filter((item) => item.danger)];
  return (
    <header className="flex flex-col gap-2">
      <div className="flex items-start gap-3">
        <div className="min-w-0 flex-1">
          {heading ?? (
            <h1 ref={titleRef} tabIndex={titleRef ? -1 : undefined} className={`break-words ${TYPE.display} ${FOCUS}`}>
              {title}
            </h1>
          )}
          {description ? <p className={`mt-0.5 max-w-[68ch] ${TYPE.bodyMuted}`}>{description}</p> : null}
        </div>
        {visible > 0 || items.length > 0 ? (
          <div data-part="actions" className="flex shrink-0 items-center gap-2">
            {control}
            {actions.map((action) => (
              <ActionButton key={action.key} action={action} />
            ))}
            {items.length > 0 ? (
              <Menu label={menuLabel} trigger={<Ellipsis className="h-4 w-4" aria-hidden="true" />} triggerClassName={`${ICON_BUTTON} border border-line`} items={items} />
            ) : null}
          </div>
        ) : null}
      </div>
      {meta ? (
        <div data-part="meta" className={`flex min-w-0 flex-wrap items-center gap-x-1.5 gap-y-1 ${TYPE.bodyMuted}`}>
          {meta}
        </div>
      ) : null}
      <div role="status" aria-live="polite" data-part="status">
        {status}
      </div>
      {children}
    </header>
  );
}
```

```ts
import type { ListGrid } from '../components/ui';

// Test-only arithmetic for a list's grid (the Records and Ops reviews: a
// container-query grid needs a summed-minimum test).

/** The grid template's tracks in rem: a minmax counts its first value. */
export function trackRems(grid: ListGrid): number[] {
  const template = /grid-cols-\[([^\]]+)\]/.exec(grid.columns)?.[1];
  if (!template) throw new Error(`no grid template in ${grid.columns}`);
  return template.split('_').map((track) => {
    const min = /^minmax\(([\d.]+)rem,/.exec(track);
    if (min) return Number(min[1]);
    const fixed = /^([\d.]+)rem$/.exec(track);
    if (fixed) return Number(fixed[1]);
    throw new Error(`unknown track ${track}`);
  });
}

/** The narrowest container a row fits: its tracks, the gaps between them
 *  and the row's own px-4 padding (2rem). */
export function minimumRem(grid: ListGrid): number {
  const tracks = trackRems(grid);
  return tracks.reduce((sum, track) => sum + track, 0) + (tracks.length - 1) * grid.gapRem + 2;
}

/** An Impact list's column at 1440: 1440px less the 72px sidebar and the
 *  frame's two 16px gutters (no Impact page has an Ask rail), in rem. */
export const IMPACT_COLUMN_AT_1440_REM = (1440 - 72 - 32) / 16;
```

```tsx
import { KIND_LABEL } from '../../features/segments/segmentFields';
import type { DayMoves, SegmentListRow } from '../../features/segments/segmentTypes';
import { movesText, OWNER_ONLY } from '../../features/segments/summaryFigures';
import { ListRow, NO_VALUE, NUM, TYPE } from '../ui';
import { SEGMENT_GRID } from './segmentGrid';
import { SizeSparkline } from './SizeSparkline';

const SHARING = { private: null, workspace: 'Workspace', people: 'Shared' } as const;

/** The no-value mark where the figure is the owner's alone (Ruling S8). */
function OwnerOnly() {
  return (
    <>
      <span aria-hidden="true">{NO_VALUE}</span>
      <span className="sr-only">{OWNER_ONLY}</span>
    </>
  );
}

/** Today's moves: "+2 / −1" in tabular figures, or the words "No change". */
function Today({ moves }: { moves: DayMoves }) {
  if (moves.entered === 0 && moves.left === 0) return <>No change</>;
  return <span className={NUM}>{movesText(moves)}</span>;
}

/** One segment (spec §3; UI foundations spec §4.2): the name opens it;
 *  kind, owner, sharing and Paused are the meta line's words; members,
 *  today's moves and the 30-day size sit in the Segments grid's columns
 *  from a 40rem list, and below it the meta line says the two figures. */
export function SegmentRow({ row }: { row: SegmentListRow }) {
  const sharing = SHARING[row.sharing];
  const members = row.member_count === null ? <OwnerOnly /> : <span className={NUM}>{row.member_count}</span>;
  const today = row.today === null ? <OwnerOnly /> : <Today moves={row.today} />;
  return (
    <ListRow
      dataAttrs={{ 'data-segment': row.id }}
      title={row.name}
      href={`/segments/${row.id}`}
      columns={SEGMENT_GRID.columns}
      gridAt={SEGMENT_GRID.gridAt}
      meta={
        <>
          {KIND_LABEL[row.kind]} · {row.is_owner ? 'You' : row.owner.name}
          {sharing ? ` · ${sharing}` : ''}
          {row.paused ? ' · Paused' : ''}
          <span className={SEGMENT_GRID.narrowOnly}>
            {' · '}
            {members} members · {today} today
          </span>
        </>
      }
      cells={[
        <span key="count" data-part="count" className={`block text-right ${TYPE.body} ${NUM}`}>
          {members}
        </span>,
        <span key="today" data-part="today" className={`block text-right ${TYPE.body}`}>
          {today}
        </span>,
        <span key="sparkline" data-part="sparkline" className="flex items-center">
          {row.sparkline === null ? <OwnerOnly /> : <SizeSparkline sizes={row.sparkline} />}
        </span>,
      ]}
    />
  );
}
```

```tsx
import type { SegmentKind, SegmentSummary } from '../../features/segments/segmentTypes';
import { summaryFigures } from '../../features/segments/summaryFigures';
import { Card, ErrorState, LoadingState, Tile, tileRow } from '../ui';

/** The tiles (spec §3; UI foundations spec §4.3) over every member the
 *  reader may open; a search on the Members tab narrows the rows, never
 *  these. One anatomy each, n-up once the page's @container is 50rem; a
 *  sideways strip on phones (the page never scrolls sideways). */
export function SegmentTiles({ summary, kind, failed, onRetry }: { summary: SegmentSummary | null; kind: SegmentKind; failed: boolean; onRetry: () => void }) {
  if (failed) {
    return (
      <Card pad="none">
        <ErrorState message="Could not load the totals." detail="The members below are not affected." onRetry={onRetry} />
      </Card>
    );
  }
  if (!summary) return <LoadingState label="Loading the totals" variant="tiles" count={kind === 'contact' ? 2 : 5} />;
  const figures = summaryFigures(summary, kind);
  const columns = Math.min(5, Math.max(2, figures.length)) as 2 | 3 | 4 | 5;
  return (
    <div className={tileRow(columns, 'container')}>
      {figures.map((figure) => (
        <Tile key={figure.key} label={figure.label} value={figure.value} meta={figure.detail ?? undefined} />
      ))}
    </div>
  );
}
```

```tsx
const SHARING = { private: null, workspace: 'Workspace', people: 'Shared' } as const;

 *  segment, date, sender and sharing are the meta line; the type, answered
 *  of sent and the rate sit in the sends grid's columns from a 40rem list
 *  (a draft shows Draft there), and below it the meta line says them. */
export function SendRow({ row }: { row: SendListRow }) {
  const draft = row.status === 'draft';
  const sharing = SHARING[row.sharing];
  const when = draft ? `Edited ${dayLabel(row.updated_at)}` : row.sent_at ? `Sent ${dayLabel(row.sent_at)}` : 'Sending now';
  const answered = draft ? null : (
    <>
      <span className={NUM}>{row.answered}</span> of <span className={NUM}>{row.sent}</span>
    </>
  );
  return (
    <ListRow
      dataAttrs={{ 'data-send': row.id }}
      title={row.name}
      href={`/surveys/${row.id}`}
      columns={SEND_GRID.columns}
      gridAt={SEND_GRID.gridAt}
      meta={
        <>
          {segmentLabel(row.segment)} · {when} · {row.is_owner ? 'You' : row.owner.name}
          {sharing ? ` · ${sharing}` : ''}
          <span className={SEND_GRID.narrowOnly}>
            {' · '}
            {TYPE_LABEL[row.survey_type]} · {draft ? 'Draft' : <>{answered} answered</>}
          </span>
        </>
      }
      cells={[
        <span key="type" className={`block ${TYPE.bodyMuted}`}>
          {TYPE_LABEL[row.survey_type]}
        </span>,
        <span key="answered" data-part="answered" className={`block text-right ${TYPE.body}`}>
          {draft ? <Badge>Draft</Badge> : answered}
        </span>,
        <span key="rate" className={`block text-right ${TYPE.body} ${NUM}`}>
          {draft ? NO_VALUE : <Rate rate={row.rate} isOwner={row.is_owner} />}
        </span>,
      ]}
    />
  );
}
```
